% ===== S4 diagnosis =====
\begin{frame}{诊断:迭代数受条件数支配,$\kappa$ 含两个病灶}
\vspace{0.3em}
\begin{center}
$n_{\text{iter}} \ \sim\ \tfrac{1}{2}\sqrt{\kappa}\,\ln(2/\varepsilon),\qquad \kappa = \kappa(M^{-1}A)$
\end{center}
\vspace{0.6em}
\[
\kappa(M^{-1}A)\ \lesssim\
\underbrace{C_0^{2}(\delta)}_{\substack{\textcolor{Teal}{\text{稳定分解}}\\[0.2em]\textcolor{Muted}{\text{overlap }\delta\uparrow\text{ 时 }\downarrow}}}
\ \cdot\
\underbrace{N_c}_{\substack{\textcolor{AnomRed}{\text{病灶 B}}\\[0.2em]\textcolor{Muted}{\text{重叠重复计数}}}}
\ \cdot\
\underbrace{\omega_{\max}/\omega_{\min}}_{\substack{\textcolor{AnomRed}{\text{病灶 A}}\\[0.2em]\textcolor{Muted}{\text{局部解不精确}}}}
\]
\vspace{0.4em}
\begin{beamercolorbox}[wd=\textwidth,sep=8pt]{}
\colorbox{NavyDark}{\parbox{0.95\textwidth}{\color{white}%
\textbf{\color{Ice}核心命题}\quad 当病灶 A 与 B 同时存在,$\delta$ 增大使 $(N_c\cdot\omega)$ 的放大盖过 $C_0^2$ 的下降 $\Rightarrow \kappa\uparrow \Rightarrow$ 迭代 $\uparrow$。\\[0.3em]
{\footnotesize\color{Ice}下面用实验逐一拆解:拔掉任一因子,趋势即翻回下降。}}}
\end{beamercolorbox}
\end{frame}
