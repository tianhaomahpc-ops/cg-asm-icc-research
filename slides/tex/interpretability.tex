% ===== 4. Concept: what is the subdomain solve (MOVED EARLY) =====
\begin{frame}{First, what is an ``exact / inexact subdomain solve''?}
\framesubtitle{$A_i=R_iAR_i^{\top}$ --- the global Laplace restricted to subdomain $i$ (with overlap)}
\begin{itemize}\itemsep0.3em
  \item $A_i$ implicitly imposes \textbf{homogeneous Dirichlet} on the \emph{artificial cut} between subdomains (cross-interface coupling dropped), keeping the physical BC on faces touching the real boundary.
  \item \textbf{Exact solve} $A_i^{-1}$ (Cholesky) $=$ \emph{exactly} solving that local Dirichlet--Laplace subproblem. The inexactness factor $\omega=1$.
  \item \textbf{Inexact solve} ICC(0) $=$ exact solve of a \emph{perturbed} operator $\tilde A_i=A_i-E_i$, where $E_i$ is the fill ICC(0) \emph{drops}. This $E_i$ \textbf{is} the source of $\omega>1$, and its error sits in the overlap/seam.
\end{itemize}
\vskip0.2em
\begin{block}{So ``inexact'' is not mysterious}
\footnotesize ICC(0) keeps only $A_i$'s nonzero pattern and throws away all fill-in. Whether that loses anything depends entirely on the \textbf{ordering} of the unknowns --- demonstrated later by reordering (Fig.~7). The rigorous measure is $\omega=\lambda_{\max}(M_i^{-1}A_i)$, the local condition factor.
\end{block}
\end{frame}

% ===== 5. Fig 1: over-counting (factor B) =====
\begin{frame}{Mechanism: BASIC over-counts the overlap (the factor $\hat N$)}
\framesubtitle{1D Laplace, 256 DOF, 8 subdomains, \emph{exact} Cholesky; one apply $\sum_i R_i^{\top}A_i^{-1}R_i\,r$ to a localized residual}
\begin{center}\includegraphics[width=0.69\textwidth]{fig1_overcount}\end{center}
\vskip-0.6em
\begin{itemize}\itemsep0.12em\scriptsize
  \item In the overlap a DOF belongs to $m_k$ subdomains, so its correction is \emph{added} $m_k$ times.
  \item \textbf{BASIC} (red) bumps up at every overlap seam; \textbf{sASM} (blue) is normalized by $1/\sqrt{m_k}$.
  \item This over-count is the algebraic fact $\sum_i R_i^{\top}R_i=\mathbf D=\mathrm{diag}(m_k)$; the max multiplicity is $\hat N=\max_k m_k$. It is harmless \emph{by itself} (note: exact solve here) --- the next slides show when it bites.
\end{itemize}
\end{frame}

% ===== 6. Q1: the bound (assertive title) =====
\begin{frame}{Q1: inexactness bites because BASIC \emph{multiplies} it by the over-count}
\framesubtitle{the abstract additive-Schwarz bound: the two factors multiply in $\lambda_{\max}$}
\[
\kappa(\mathbf M^{-1}\mathbf A)\ \le\ C_0^2\ \cdot\
\underbrace{\omega}_{\substack{\text{(A) inexactness}\\ \text{exact}=1,\ \text{ICC}>1}}\ \cdot\
\underbrace{\Nc}_{\substack{\text{(B) over-count}\\ \hat N=\max_k m_k\,\le\,\Nc}}
\]
\begin{itemize}\itemsep0.16em
  \item Over-count factor: $\mathbf D=\mathrm{diag}(m_k)$ measures the \textbf{max multiplicity} $\hat N$; rigorously $\hat N\le\Nc$ (Gander et al.\ 2015, Rem.~2.8). It is what the scaling removes.
  \item \textbf{Exact} ($\omega{=}1$): corrections are exact; over-counting only lifts $\lambda_{\max}\!\le\!\hat N$ --- bounded, CG tolerates it. \emph{No anomaly} (23 vs 27).
  \item \textbf{ICC} ($\omega{>}1$): each correction carries an \emph{error}; BASIC \textbf{adds $m_k$ of them} in the overlap $\Rightarrow$ $\lambda_{\max}\!\approx\!\omega\,\hat N$, inexactness \emph{times} over-count.
  \item Wider overlap: $\omega\!\uparrow$ \emph{and} $\hat N\!\uparrow$ $\Rightarrow$ $\kappa$ \emph{rises} with overlap (the anomaly); the amplified error \textbf{piles up at the seams}.
\end{itemize}
\end{frame}

% ===== 10. Fig 7 reading + pin omega =====
\begin{frame}{Only band-widening reorderings break exactness; sASM repairs them}
\framesubtitle{$\eta$ (cheap $\ell_2$ residual flag) and the rigorous $\kappa(M_i^{-1}A_i)=\lambda_{\max}/\lambda_{\min}$ track together; recall $\omega=\lambda_{\max}(M_i^{-1}A_i)$}
\begin{itemize}\itemsep0.22em\footnotesize
  \item \textbf{(b) the headline.} 1D \emph{natural} \& \emph{RCM}: $\eta\approx10^{-15}$, $\kappa=1$ --- ICC \emph{is} exact Cholesky (tridiagonal has \emph{no fill}). 1D \emph{random}: $\eta=0.67$, $\kappa=89$ --- the permutation creates fill ICC(0) throws away, so ICC is now inexact. ($\eta$ is only a flag; $\kappa$, the green line, is the rigorous condition number $=\omega$.)
  \item \textbf{(a)/(d) 1D.} natural/RCM: 15; random blows up to 113 --- yet \textbf{sASM barely helps} (113$\to$108), because 1D's over-count is only $\hat N{=}2$: A ($\kappa{=}89$) is large but B ($\hat N$) is small, so there is little to remove. (RCM stays banded $\Rightarrow$ exact: not every permutation breaks it.)
  \item \textbf{(c)/(d) 2D.} The 5-point ICC already has fill ($\kappa{=}15$); random makes it worse ($\kappa{=}28$): $102\!\to\!147$ it. Here $\hat N{=}4$, so \textbf{sASM repairs both} ($102\!\to\!67$, $147\!\to\!92$). \emph{The sASM win needs both A and B large} --- 1D random (A only) vs 2D/3D.
\end{itemize}

% ===== 12. Q2: why sASM fixes it =====
\begin{frame}{Q2: the $\mathbf D^{-1/2}$ partition of unity removes $\hat N$, so $\lambda_{\max}\le\omega$}
\framesubtitle{$\mathbf M_{\mathrm{sASM}}^{-1}=\mathbf D^{-1/2}\big(\textstyle\sum_i R_i^{\top}A_i^{-1}R_i\big)\mathbf D^{-1/2}$, partition of unity $\sum_i d_i^2=1$}
\[
\text{BASIC: } \lambda_{\max}\le \omega\,\hat N
\qquad\xrightarrow{\ \mathbf D^{-1/2}(\cdot)\mathbf D^{-1/2}\ }\qquad
\text{sASM: } \lambda_{\max}\le \omega
\]
\begin{itemize}\itemsep0.3em
  \item The $\mathbf D^{-1/2}$ sandwich restores a partition of unity ($\sum_i$ scaled $R_i^{\top}R_i=I$), so each DOF is counted \textbf{once}: the multiplicity over-count $\hat N$ is \textbf{removed} (Fig.~1 blue flat).
  \item \textbf{This breaks the multiplication.} $\omega$ is no longer multiplied by $\hat N$: the ICC error in the overlap is counted once, \emph{not accumulated} $\Rightarrow$ no seam pile-up; CG recovers \textbf{102 $\to$ 67}.
  \item \textbf{sASM does not fix the inexactness} ($\omega$ still $>1$) --- it stops the over-count from \emph{amplifying} it. So $\kappa\lesssim C_0^2\,\omega$, and overlap helps again. Measured next.
\end{itemize}
\end{frame}

% ===== 16. Conclusion =====
\begin{frame}{Conclusion: over-count multiplies inexactness; the partition of unity breaks it}
\begin{block}{The one-sentence picture}
\footnotesize
BASIC counts the overlap $m_k$ times (Fig.~1). With \emph{exact} solves that is harmless --- it only lifts
$\lambda_{\max}\!\le\!\hat N$, which CG tolerates (Fig.~3, solid). With an \emph{inexact} ICC solve the local error is
counted $m_k$ times, so ``inexactness $\times$ over-count'' ($\omega\hat N$) accumulates at the seams (Fig.~4) and convergence
degrades (Fig.~3, dashed, 102). sASM's $\mathbf D^{-1/2}$ partition of unity removes the multiplicity over-count, breaking
the product, so the inexact corrections are no longer amplified and CG recovers (102 $\to$ 67).
\end{block}
\vskip0.3em
\begin{itemize}\itemsep0.26em\footnotesize
  \item \textbf{Q1:} inexactness is bad because BASIC \emph{multiplies} it by $\hat N=\max_k m_k$; both grow with overlap (the anomaly). Measured: $\lambda_{\max}(\text{BASIC,exact})=\hat N$, lifted to $\omega\hat N$ by ICC (Fig.~8).
  \item \textbf{Q2:} sASM removes $\hat N$ (partition of unity, Fig.~8 flat), so it helps \emph{exactly} when the solve is inexact, and is neutral when exact.
  \item \textbf{Second thread:} information propagates at finite, overlap-set speed (Fig.~2/10) $=$ the small $\lambda_{\min}$; a symmetric \textbf{two-level sASM} (coarse space) lifts it, halving iterations and making them \emph{scalable} (Fig.~9).
\end{itemize}
\end{frame}
